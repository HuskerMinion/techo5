// TECHO5 adaptation: all Soloist invocations, including update validation, use the private loader.
package main

import (
    "context"
    "os"
    "os/exec"
)

func soloistCommand(ctx context.Context, binary string, args ...string) *exec.Cmd {
    if launcher := os.Getenv("SOLOIST_EXEC"); launcher != "" {
        return exec.CommandContext(ctx, launcher, append([]string{binary}, args...)...)
    }
    return exec.CommandContext(ctx, binary, args...)
}
